updateGrid(gridList) {
let newGridList = [];
for ( let r = 0; r < height; r++ ) {
for ( let c = 0; c < height; c++) {
let neighbourCount = getNeighbourCount(r, c);
let currentStatus = gridList[r][c];
let newState = false;
// A living cell with less than two neighbours: dies!
if(currentStatus && neighbourCount < 2) {
newState = false;
}
// A living cell with two or three neighbours: survives!
if(neighbourCount === 3 || (currentStatus && neigbourCount === 2) {
newState = true;
}
// A living cell with more than 3 neighbours: dies!
if(currentStatus && neighbourCount >= 3) {
newState = false;
}
// A dead cell with exactly 3 neighbours: becomes alive
if(!currentStatus && neighbourCount === 3) {
newState = true;
}
newGridList[r][c] = newState;
}
}
return newGridList;
}
function getNeighbourCount(row, column) {
if(row > height || column > width) {
return 0;
}
let neighbours = 0;
for ( let offsetX = column - 1; offsetX <= column + 1; offsetX++ ) {
for (let offsetY = row - 1; offsetY <= row + 1; offsetY++) {
// If the current neightbour is the current cell, continue
if(offsetX == column && offsetY == row) {
continue;
}
// If the current neighbour is out of bounds, we continue to the next
if(offsetX <= 0 || offsetY <= 0 || offsetX > width || offsetY > height) {
continue;
}
if(gridList[offsetY][offsetX] === true) {
neighbours++;
}
}
}
return neighbours;
}
